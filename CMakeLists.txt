cmake_minimum_required(VERSION 3.20)
project(shoutout-desk-obs VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This release supports Windows x64 only.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
add_library(shoutout-desk-obs MODULE src/plugin.cpp src/dock.cpp)
target_link_libraries(shoutout-desk-obs PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
target_compile_definitions(shoutout-desk-obs PRIVATE NOMINMAX)
add_executable(shoutout-secure src/secure.cpp)
target_link_libraries(shoutout-secure PRIVATE Crypt32)
add_executable(shoutout-preview WIN32 src/preview.cpp src/dock.cpp)
target_link_libraries(shoutout-preview PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
foreach(target shoutout-desk-obs shoutout-secure shoutout-preview)
  target_compile_options(${target} PRIVATE /utf-8 /W4 /permissive- /Zc:__cplusplus)
endforeach()
install(TARGETS shoutout-desk-obs RUNTIME DESTINATION bin/64bit LIBRARY DESTINATION bin/64bit)
